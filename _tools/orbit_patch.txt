  // ═══ ★ 2026-11-01 新：选中后「其它星球绕它公转」═══
  // 用户原话：「不再是三点一线，而是将选中的星球当成『恒星』，其它星球绕其旋转
  //            （更方便观测其与周边星球的联系）」
  //
  // 原实现的问题：pushOut 沿 **深度方向** 推远（depth + pushOut*2.0），
  // 投影后所有被推的节点都沿同一条视线排开 —— 就是用户看到的「三点一线」。
  // 而且深度推远不改变屏幕上的角度，所以「谁和谁挨着」根本看不出来。
  //
  // 改法：在**屏幕空间**做极坐标轨道。焦点固定在中心，其余节点按关系强度分配半径：
  //   强关联（fw 高）→ 近轨道，转得慢
  //   弱关联（fw 低）→ 远轨道，转得快
  // 这样一眼就能看出「谁离它近、谁离它远」，而距离本身就是关系强度。
  let outSx = viewCx + cx1 * persp * screenScale;
  let outSy = viewCy + cy2 * persp * screenScale;
  if (orbitLevel > 0.02 && focusNode && p !== focusNode) {
    const fw = fwOf(p);
    // 焦点在屏幕上的位置：相机已对准它，所以基本上就是视图锚点。
    // 用上一帧缓存更稳（相机飞行途中也能跟上）。
    const fx = _focusSX || viewCx, fy = _focusSY || viewCy;
    let dx = outSx - fx, dy = outSy - fy;
    let dist = Math.hypot(dx, dy);
    const ang0 = dist < 0.6
      ? idPhase(p.id) * 6.283
      : Math.atan2(dy, dx);
    if (dist < 0.6) dist = 1;
    const baseR = Math.min(W, H) * 0.30;
    // 半径由关系强度决定 —— 这是「轨道」的全部意义
    const targetR = fw >= 0.5
      ? baseR * (0.36 + (1 - fw) * 0.5)      // 邻居：贴着转
      : baseR * (0.70 + (1 - fw) * 1.5);     // 无关：外圈
    // 角速度：近轨道慢、远轨道快（开普勒式的观感，不是物理公式）
    const spinAng = idPhase(p.id) * 6.283 + orbitAngle * (1.0 + (1 - fw) * 2.2);
    const ang = ang0 + (spinAng - ang0) * orbitLevel;
    const r = dist + (targetR - dist) * orbitLevel;
    outSx = fx + Math.cos(ang) * r;
    outSy = fy + Math.sin(ang) * r;
  }
  return {
    sx: outSx,
    sy: outSy,
    s: persp * zoom,
    persp,
    depth,
    z: depth,
    behind: false,
  };
}
